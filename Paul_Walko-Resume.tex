% Custom commands
\newcommand{\resumeItem}[1]{
  \item\small{
    {#1 \vspace{-2pt}}
  }
}

% #1 job title, #2 employer, #3 location, #4 dates
%
% Three left-aligned lines, one field per line where it matters. Workday was
% observed to take "everything before the date range on the line" as the
% employer, so putting the employer on a line by itself -- and leading the
% third line with the dates -- keeps the location out of the employer field.
\newcommand{\resumeSubheading}[4]{
  \vspace{-2pt}\item
    \textbf{#1} \\
    {\small\textit{#2}} \\
    {\small\textit{#4} \sep{} \textit{#3}}
    \vspace{-5pt}
}

% #1 heading line, including any dates, separated with \sep
\newcommand{\resumeProjectHeading}[1]{
    \item
    \small{#1}\vspace{-5pt}
}

\newcommand{\resumeSubItem}[1]{\resumeItem{#1}\vspace{-4pt}}

% #1 school, #2 degree, #3 years, #4 location
% Education parsers key off the school name first and want bare years, not
% "August 2015 - December 2018" -- Workday returned no years for that form.
\newcommand{\resumeEducation}[4]{
  \vspace{-2pt}\item
    \textbf{#1} \\
    {\small\textit{#2}} \\
    {\small\textit{#3} \sep{} \textit{#4}}
    \vspace{-5pt}
}

% Inline separator. Kept as a macro so the docx build can swap the math-mode
% bar for a literal "|" -- inline math becomes an unreadable Word equation.
\newcommand{\sep}{$|$}

% Plain text lines (Skills, Talks). Separate from \resumeSubHeadingListStart so
% the docx build can render entry lists as paragraphs and these as bullets.
\newcommand{\resumeTextListStart}{\begin{itemize}[leftmargin=0.15in, label={}]}
\newcommand{\resumeTextListEnd}{\end{itemize}}
\newcommand{\resumeTextItem}[1]{\item{\small #1}\vspace{-4pt}}

\renewcommand\labelitemii{$\vcenter{\hbox{\tiny$\bullet$}}$}

\newcommand{\resumeSubHeadingListStart}{\begin{itemize}[leftmargin=0.15in, label={}]}
\newcommand{\resumeSubHeadingListEnd}{\end{itemize}}
\newcommand{\resumeItemListStart}{\begin{itemize}}
\newcommand{\resumeItemListEnd}{\end{itemize}\vspace{-5pt}}
